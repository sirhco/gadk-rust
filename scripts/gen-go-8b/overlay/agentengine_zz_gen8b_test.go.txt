// 8b overlay probe: writes the gadk-cli Agent Engine environment goldens
// (crates/gadk-cli/testdata/environment/) with ADK-Go's cmd/adkgo/internal/deploy/agentengine
// itself at docs/parity/ADK_GO_COMMIT. Not a file of this repository's build (hence the .txt
// suffix): scripts/gen-go-8b/run.sh maps it into that package as zz_gen8b_test.go with
// `go test -overlay` and runs `go test -run '^TestZZGen8bEnvironment$'` with GEN_GO_8B_OUT set.
// ADK-Go's tree is not modified.
//
// The scenarios (name, args, update) are read from the TestEnvironmentDeploymentRequests table in
// environment_test.go itself (go/ast), not transcribed. Each one runs exactly as that test does:
// the real cobra command (environmentCommand) up to its bounded entry-point error, then
// computeFlags with entry point main.go, a "test archive" source archive, and
// gcloudDeployToAgentEngine / gcloudUpdateAgentEngine against the package's own loopback gRPC
// server (environmentDeploymentServer / environmentClientOptions), which captures the request.
// The only rewrite is the create request's clock-derived default display name
// ("ADK Agent: <RFC3339 now>"), recorded as "ADK Agent: {{RFC3339}}".
//
// Every identifier here starts with zz so that it cannot collide with the package or its tests.

package agentengine

import (
	"bytes"
	"encoding/json"
	"go/ast"
	"go/parser"
	"go/token"
	"os"
	"path/filepath"
	"regexp"
	"strconv"
	"strings"
	"testing"

	"cloud.google.com/go/aiplatform/apiv1beta1/aiplatformpb"
	"google.golang.org/protobuf/encoding/protojson"
	"google.golang.org/protobuf/proto"
)

type zzScenario struct {
	name   string
	args   []string
	update bool
}

func zzUnquote(t *testing.T, e ast.Expr) string {
	t.Helper()
	lit, ok := e.(*ast.BasicLit)
	if !ok || lit.Kind != token.STRING {
		t.Fatalf("expected a string literal, got %T", e)
	}
	s, err := strconv.Unquote(lit.Value)
	if err != nil {
		t.Fatal(err)
	}
	return s
}

// zzScenarios reads the `tests` table of TestEnvironmentDeploymentRequests in environment_test.go.
func zzScenarios(t *testing.T) []zzScenario {
	t.Helper()
	f, err := parser.ParseFile(token.NewFileSet(), "environment_test.go", nil, 0)
	if err != nil {
		t.Fatal(err)
	}
	var out []zzScenario
	for _, decl := range f.Decls {
		fd, ok := decl.(*ast.FuncDecl)
		if !ok || fd.Name.Name != "TestEnvironmentDeploymentRequests" {
			continue
		}
		ast.Inspect(fd.Body, func(n ast.Node) bool {
			as, ok := n.(*ast.AssignStmt)
			if !ok || len(as.Lhs) != 1 || len(as.Rhs) != 1 {
				return true
			}
			if id, ok := as.Lhs[0].(*ast.Ident); !ok || id.Name != "tests" {
				return true
			}
			table, ok := as.Rhs[0].(*ast.CompositeLit)
			if !ok {
				t.Fatal("tests is not a composite literal")
			}
			for _, elt := range table.Elts {
				row, ok := elt.(*ast.CompositeLit)
				if !ok {
					t.Fatalf("unexpected table row %T", elt)
				}
				var s zzScenario
				for _, field := range row.Elts {
					kv, ok := field.(*ast.KeyValueExpr)
					if !ok {
						t.Fatalf("unkeyed table field %T", field)
					}
					switch kv.Key.(*ast.Ident).Name {
					case "name":
						s.name = zzUnquote(t, kv.Value)
					case "args":
						for _, a := range kv.Value.(*ast.CompositeLit).Elts {
							s.args = append(s.args, zzUnquote(t, a))
						}
					case "update":
						s.update = kv.Value.(*ast.Ident).Name == "true"
					}
				}
				out = append(out, s)
			}
			return false
		})
	}
	if len(out) == 0 {
		t.Fatal("no TestEnvironmentDeploymentRequests scenarios found")
	}
	return out
}

var zzNonAlnum = regexp.MustCompile(`[^a-z0-9]+`)

// zzSlug is the golden file stem of a scenario name: lower case, every run of other characters
// replaced by one underscore, no leading or trailing underscore.
func zzSlug(name string) string {
	return strings.Trim(zzNonAlnum.ReplaceAllString(strings.ToLower(name), "_"), "_")
}

// zzProtoJSON is protojson output with its deliberately unstable whitespace normalized away.
func zzProtoJSON(t *testing.T, m proto.Message) json.RawMessage {
	t.Helper()
	raw, err := protojson.Marshal(m)
	if err != nil {
		t.Fatal(err)
	}
	var buf bytes.Buffer
	if err := json.Compact(&buf, raw); err != nil {
		t.Fatal(err)
	}
	return buf.Bytes()
}

type zzGolden struct {
	Name          string          `json:"name"`
	Args          []string        `json:"args"`
	Update        bool            `json:"update"`
	ProjectName   string          `json:"projectName"`
	Region        string          `json:"region"`
	AgentEngineID string          `json:"agentEngineId,omitempty"`
	Request       json.RawMessage `json:"request"`
}

func TestZZGen8bEnvironment(t *testing.T) {
	out := os.Getenv("GEN_GO_8B_OUT")
	if out == "" {
		t.Skip("GEN_GO_8B_OUT is not set")
	}
	dir := filepath.Join(out, "environment")
	if err := os.MkdirAll(dir, 0o755); err != nil {
		t.Fatal(err)
	}
	for _, sc := range zzScenarios(t) {
		t.Run(sc.name, func(t *testing.T) {
			resetFlags(t, "not-a-go-file")
			flags.gcloud.projectName = "test-project"
			flags.gcloud.region = "test-region"
			cmd := environmentCommand()
			cmd.SetArgs(sc.args)
			g := zzGolden{Name: sc.name, Args: sc.args, Update: sc.update, ProjectName: "test-project", Region: "test-region"}
			if g.Args == nil {
				g.Args = []string{}
			}
			captureEnvironmentOutput(t, func() {
				if err := cmd.Execute(); err == nil || strings.Contains(err.Error(), "--env entry") {
					t.Fatalf("Execute() error = %v, want the bounded entry-point error", err)
				}
				flags.source.entryPointPath = "main.go"
				if err := flags.computeFlags(); err != nil {
					t.Fatal(err)
				}
				if err := os.WriteFile(flags.build.archivePath, []byte("test archive"), 0o600); err != nil {
					t.Fatal(err)
				}
				server := &environmentDeploymentServer{created: make(chan *aiplatformpb.CreateReasoningEngineRequest, 1), updated: make(chan *aiplatformpb.UpdateReasoningEngineRequest, 1)}
				opts := environmentClientOptions(t, server)
				if sc.update {
					flags.agentEngine.agentEngineID = "test-engine"
					g.AgentEngineID = "test-engine"
					if err := flags.gcloudUpdateAgentEngine(opts...); err != nil {
						t.Fatal(err)
					}
					g.Request = zzProtoJSON(t, <-server.updated)
				} else {
					if err := flags.gcloudDeployToAgentEngine(opts...); err != nil {
						t.Fatal(err)
					}
					req := <-server.created
					if dn := req.GetReasoningEngine().GetDisplayName(); !strings.HasPrefix(dn, "ADK Agent: ") {
						t.Fatalf("unexpected display name %q", dn)
					}
					req.ReasoningEngine.DisplayName = "ADK Agent: {{RFC3339}}"
					g.Request = zzProtoJSON(t, req)
				}
			})
			var buf bytes.Buffer
			enc := json.NewEncoder(&buf)
			enc.SetEscapeHTML(false)
			enc.SetIndent("", "  ")
			if err := enc.Encode(g); err != nil {
				t.Fatal(err)
			}
			file := filepath.Join(dir, zzSlug(sc.name)+".json")
			if err := os.WriteFile(file, buf.Bytes(), 0o644); err != nil {
				t.Fatal(err)
			}
		})
	}
}
