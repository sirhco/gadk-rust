// 8b overlay probe: writes the gadk-vertexai Example Store goldens
// (crates/gadk-vertexai/testdata/example_store/) with ADK-Go's tool/exampletool/vertexai itself at
// docs/parity/ADK_GO_COMMIT. Not a file of this repository's build (hence the .txt suffix):
// scripts/gen-go-8b/run.sh maps it into ADK-Go's tool/exampletool/vertexai package as
// zz_gen8b_test.go with `go test -overlay` and runs `go test -run '^TestZZGen8bExampleStore$'` with
// GEN_GO_8B_OUT set. ADK-Go's tree is not modified.
//
// Each case below is the input of the TestProvider_Examples subtest of the same name in
// vertexai_test.go (query, fake client response or error), transcribed verbatim; the test fails
// unless the case names equal that function's t.Run names, so a Go-side change cannot be missed.
// The request is what Go's provider sends (captured by the package's own fakeExampleStoreClient),
// the expected output is what Go's provider returns for the response.
//
// Every identifier here starts with zz so that it cannot collide with the package or its tests.

type zzCase struct {
	name      string
	query     string
	resp      *aiplatformpb.SearchExamplesResponse
	searchErr error
}

func zzMustStruct(t *testing.T, m map[string]any) *structpb.Struct {
	t.Helper()
	s, err := structpb.NewStruct(m)
	if err != nil {
		t.Fatal(err)
	}
	return s
}

func zzText(text string) *aiplatformpb.Part {
	return &aiplatformpb.Part{Data: &aiplatformpb.Part_Text{Text: text}}
}

// zzSubtestNames returns the t.Run names of the named test function in file.
func zzSubtestNames(t *testing.T, file, fn string) []string {
	t.Helper()
	f, err := parser.ParseFile(token.NewFileSet(), file, nil, 0)
	if err != nil {
		t.Fatal(err)
	}
	var names []string
	for _, decl := range f.Decls {
		fd, ok := decl.(*ast.FuncDecl)
		if !ok || fd.Name.Name != fn {
			continue
		}
		ast.Inspect(fd.Body, func(n ast.Node) bool {
			call, ok := n.(*ast.CallExpr)
			if !ok || len(call.Args) != 2 {
				return true
			}
			sel, ok := call.Fun.(*ast.SelectorExpr)
			if !ok || sel.Sel.Name != "Run" {
				return true
			}
			if lit, ok := call.Args[0].(*ast.BasicLit); ok && lit.Kind == token.STRING {
				s, err := strconv.Unquote(lit.Value)
				if err != nil {
					t.Fatal(err)
				}
				names = append(names, s)
			}
			return true
		})
	}
	return names
}

var zzNonAlnum = regexp.MustCompile(`[^a-z0-9]+`)

// zzSlug is the golden file stem of a subtest name: lower case, every run of other characters
// replaced by one underscore, no leading or trailing underscore.
func zzSlug(name string) string {
	return strings.Trim(zzNonAlnum.ReplaceAllString(strings.ToLower(name), "_"), "_")
}

// zzProtoJSON is protojson output with its deliberately unstable whitespace normalized away.
func zzProtoJSON(t *testing.T, opts protojson.MarshalOptions, m proto.Message) json.RawMessage {
	t.Helper()
	raw, err := opts.Marshal(m)
	if err != nil {
		t.Fatal(err)
	}
	var buf bytes.Buffer
	if err := json.Compact(&buf, raw); err != nil {
		t.Fatal(err)
	}
	return buf.Bytes()
}

type zzGolden struct {
	Name          string          `json:"name"`
	Query         string          `json:"query"`
	Request       json.RawMessage `json:"request"`
	Response      json.RawMessage `json:"response,omitempty"`
	ResponseIsNil bool            `json:"responseIsNil,omitempty"`
	SearchError   string          `json:"searchError,omitempty"`
	Examples      json.RawMessage `json:"examples,omitempty"`
	Error         string          `json:"error,omitempty"`
}

func TestZZGen8bExampleStore(t *testing.T) {
	out := os.Getenv("GEN_GO_8B_OUT")
	if out == "" {
		t.Skip("GEN_GO_8B_OUT is not set")
	}
	cases := zzCases(t)
	var names []string
	for _, c := range cases {
		names = append(names, c.name)
	}
	if goNames := zzSubtestNames(t, "vertexai_test.go", "TestProvider_Examples"); !slices.Equal(names, goNames) {
		t.Fatalf("case names differ from TestProvider_Examples' subtests:\n got %q\nwant %q", names, goNames)
	}
	dir := filepath.Join(out, "example_store")
	if err := os.MkdirAll(dir, 0o755); err != nil {
		t.Fatal(err)
	}
	for _, c := range cases {
		client := &fakeExampleStoreClient{resp: c.resp, err: c.searchErr}
		p, err := newProvider(&ProviderConfig{ExampleStoreName: testStoreName}, client)
		if err != nil {
			t.Fatal(err)
		}
		examples, err := p.Examples(t.Context(), c.query)
		g := zzGolden{
			Name:    c.name,
			Query:   c.query,
			Request: zzProtoJSON(t, protojson.MarshalOptions{}, client.gotReq),
		}
		switch {
		case c.searchErr != nil:
			g.SearchError = c.searchErr.Error()
		case c.resp == nil:
			g.ResponseIsNil = true
			g.Response = json.RawMessage("{}")
		default:
			g.Response = zzProtoJSON(t, protojson.MarshalOptions{UseEnumNumbers: true}, c.resp)
		}
		if err != nil {
			g.Error = err.Error()
		} else {
			raw, merr := json.Marshal(examples)
			if merr != nil {
				t.Fatal(merr)
			}
			g.Examples = raw
		}
		var buf bytes.Buffer
		enc := json.NewEncoder(&buf)
		enc.SetEscapeHTML(false)
		enc.SetIndent("", "  ")
		if err := enc.Encode(g); err != nil {
			t.Fatal(err)
		}
		file := filepath.Join(dir, zzSlug(c.name)+".json")
		if err := os.WriteFile(file, buf.Bytes(), 0o644); err != nil {
			t.Fatal(err)
		}
		t.Logf("wrote %s", file)
	}
}
