// (crates/gadk-agentanalytics/tests/fixtures/) with ADK-Go's plugin/agentanalytics itself at
// docs/parity/ADK_GO_COMMIT. Not a file of this repository's build (hence the .txt suffix):

const zzOutEnv = "AGENTANALYTICS_PROBE_OUT"

// zzMaxInlineOutput is the longest SmartTruncate output written inline in a fixture; longer
// outputs are recorded by length, SHA-256 and their first and last 256 bytes.
const zzMaxInlineOutput = 64 * 1024

type zzProbe struct {
	t     *testing.T
	root  string
	files []string
}

func TestGenerateFixtures(t *testing.T) {
	out := os.Getenv(zzOutEnv)
	if out == "" {
		t.Skip(zzOutEnv + " not set")
	}
	p := &zzProbe{t: t, root: out}
	p.genConfig()
	p.genSchema()
	p.genArrow()
	p.genViews()
	p.genSmartTruncate()
	p.genContent()
	p.genPartsCollector()
	p.genSummaryBuilder()
	p.genNormalizeFunctionResponse()
	p.genRedaction()
	p.genWire()
	sort.Strings(p.files)
	t.Logf("wrote %d fixture files to %s", len(p.files), out)
	fmt.Printf("wrote %d fixture files\n", len(p.files))
}

// zzMarshal is json.Marshal without HTML escaping and without a trailing newline. Map keys are
// sorted (encoding/json).
func zzMarshal(v any) []byte {
	var buf bytes.Buffer
	enc := json.NewEncoder(&buf)
	enc.SetEscapeHTML(false)
	if err := enc.Encode(v); err != nil {
		panic(fmt.Sprintf("zzMarshal: %v", err))
	}
	return bytes.TrimSuffix(buf.Bytes(), []byte("\n"))
}

func (p *zzProbe) writeFile(rel string, data []byte) {
	p.t.Helper()
	path := filepath.Join(p.root, filepath.FromSlash(rel))
	if err := os.MkdirAll(filepath.Dir(path), 0o755); err != nil {
		p.t.Fatal(err)
	}
	if err := os.WriteFile(path, data, 0o644); err != nil {
		p.t.Fatal(err)
	}
	p.files = append(p.files, rel)
}

// writeJSON writes v with 2-space indentation, no HTML escaping and a trailing newline.
func (p *zzProbe) writeJSON(rel string, v any) {
	p.t.Helper()
	var buf bytes.Buffer
	enc := json.NewEncoder(&buf)
	enc.SetEscapeHTML(false)
	enc.SetIndent("", "  ")
	if err := enc.Encode(v); err != nil {
		p.t.Fatalf("%s: %v", rel, err)
	}
	p.writeFile(rel, buf.Bytes())
}

// writeCases writes {"about": about, "cases": [...]} with one compact case per line.
func (p *zzProbe) writeCases(rel, about string, cases []any) {
	p.t.Helper()
	var buf bytes.Buffer
	buf.WriteString("{\n  \"about\": ")
	buf.Write(zzMarshal(about))
	buf.WriteString(",\n  \"cases\": [")
	for i, c := range cases {
		if i > 0 {
			buf.WriteString(",")
		}
		buf.WriteString("\n    ")
		buf.Write(zzMarshal(c))
	}
	buf.WriteString("\n  ]\n}\n")
	if !json.Valid(buf.Bytes()) {
		p.t.Fatalf("%s: invalid JSON", rel)
	}
	p.writeFile(rel, buf.Bytes())
}

// zzStr returns s, or {"invalid_utf8_base64": ...} when s is not valid UTF-8 (JSON cannot
// carry it, and encoding/json would replace the bad bytes with U+FFFD).
func zzStr(s string) any {
	if utf8.ValidString(s) {
		return s
	}
	return map[string]any{"invalid_utf8_base64": base64.StdEncoding.EncodeToString([]byte(s))}
}

func zzErrText(err error) any {
	if err == nil {
		return nil
	}
	return zzStr(err.Error())
}

// zzOutput describes the bytes written by SmartTruncate: inline when short, by digest otherwise.
func zzOutput(b []byte) map[string]any {
	if len(b) <= zzMaxInlineOutput {
		return map[string]any{"output": zzStr(string(b))}
	}
	sum := sha256.Sum256(b)
	return map[string]any{
		"output_len":    len(b),
		"output_sha256": hex.EncodeToString(sum[:]),
		"output_head":   zzStr(string(b[:256])),
		"output_tail":   zzStr(string(b[len(b)-256:])),
	}
}

func zzNs(d time.Duration) int64 { return int64(d) }

func zzDecodeJSON(t *testing.T, s string) any {
	t.Helper()
	var v any
	if err := json.Unmarshal([]byte(s), &v); err != nil {
		t.Fatalf("zzDecodeJSON(%q): %v", s, err)
	}
	return v
}

func zzRawJSON(t *testing.T, s string) json.RawMessage {
	t.Helper()
	if !json.Valid([]byte(s)) {
		t.Fatalf("invalid JSON literal: %s", s)
	}
	return json.RawMessage(s)
}

func zzSortedKeys(m map[string]struct{}) []string {
	keys := make([]string, 0, len(m))
	for k := range m {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	return keys
}

// zzDenylists are the PayloadColumnDenylist values probed for the schema, Arrow and view
// projections: the single columns the ADK-Go tests deny, a pair, and all four.
var zzDenylists = []struct {
	name   string
	denied []string
}{
	{"none", nil},
	{"content", []string{columnContent}},
	{"content_parts", []string{columnContentParts}},
	{"attributes", []string{columnAttributes}},
	{"latency_ms", []string{columnLatencyMS}},
	{"attributes_latency_ms", []string{columnAttributes, columnLatencyMS}},
	{"all", []string{columnContent, columnContentParts, columnAttributes, columnLatencyMS}},
}

// zzRESTSchema returns schema the way the bigquery client sends it in tables.insert.
func (p *zzProbe) zzRESTSchema(schema bq.Schema) json.RawMessage {
	p.t.Helper()
	srv := newZZServer(p.t, []zzCanned{{Status: 200, Body: json.RawMessage(`{}`)}})
	defer srv.close()
	ctx := context.Background()
	client := srv.client(ctx, "schema-project")
	defer client.Close()
	if err := client.Dataset("d").Table("t").Create(ctx, &bq.TableMetadata{Schema: schema}); err != nil {
		p.t.Fatalf("capture schema: %v", err)
	}
	ex := srv.exchanges
	if len(ex) != 1 {
		p.t.Fatalf("capture schema: %d requests", len(ex))
	}
	var body map[string]json.RawMessage
	if err := json.Unmarshal(ex[0].Request.BodyRaw(), &body); err != nil {
		p.t.Fatal(err)
	}
	return body["schema"]
}

func (p *zzProbe) genSchema() {
	p.writeJSON("schema/events_schema.json", map[string]any{
		"about":  "EventsSchema() as the BigQuery REST TableSchema JSON the bigquery client sends in tables.insert.",
		"schema": p.zzRESTSchema(EventsSchema()),
	})
	var cases []any
	for _, d := range zzDenylists {
		denied, err := parsePayloadColumnDenylist(d.denied)
		if err != nil {
			p.t.Fatal(err)
		}
		cases = append(cases, map[string]any{
			"name":   d.name,
			"denied": d.denied,
			"schema": p.zzRESTSchema(projectEventsSchema(denied)),
		})
	}
	p.writeCases("schema/projections.json",
		"projectEventsSchema(parsePayloadColumnDenylist(denied)) as REST TableSchema JSON.", cases)

	var parse []any
	for _, in := range [][]string{nil, {}, {"content"}, {"content", "content"}, {"bogus"}, {"content", "Content"}, {"event_type"}, {""}} {
		denied, err := parsePayloadColumnDenylist(in)
		var set any
		if err == nil {
			set = zzSortedKeys(denied)
		}
		parse = append(parse, map[string]any{"input": in, "denied": set, "error": zzErrText(err)})
	}
	p.writeCases("schema/parse_denylist.json",
		"parsePayloadColumnDenylist(input): denied (sorted set) or error text.", parse)
}

func zzArrowType(dt arrow.DataType) map[string]any {
	m := map[string]any{"go_string": dt.String()}
	switch t := dt.(type) {
	case *arrow.TimestampType:
		m["id"] = "timestamp"
		m["unit"] = t.Unit.String()
		m["timezone"] = t.TimeZone
	case *arrow.StringType:
		m["id"] = "utf8"
	case *arrow.Int64Type:
		m["id"] = "int64"
	case *arrow.BooleanType:
		m["id"] = "bool"
	case *arrow.ListType:
		m["id"] = "list"
		m["element"] = zzArrowField(t.ElemField())
	case *arrow.StructType:
		m["id"] = "struct"
		var fields []any
		for _, f := range t.Fields() {
			fields = append(fields, zzArrowField(f))
		}
		m["fields"] = fields
	default:
		panic("unexpected arrow type " + dt.String())
	}
	return m
}

func zzArrowField(f arrow.Field) map[string]any {
	m := map[string]any{"name": f.Name, "nullable": f.Nullable, "type": zzArrowType(f.Type)}
	md := [][2]string{}
	for i, k := range f.Metadata.Keys() {
		md = append(md, [2]string{k, f.Metadata.Values()[i]})
	}
	m["metadata"] = md
	return m
}

func zzArrowSchema(s *arrow.Schema) map[string]any {
	var fields []any
	for _, f := range s.Fields() {
		fields = append(fields, zzArrowField(f))
	}
	md := [][2]string{}
	for i, k := range s.Metadata().Keys() {
		md = append(md, [2]string{k, s.Metadata().Values()[i]})
	}
	return map[string]any{"fields": fields, "metadata": md, "endianness": s.Endianness().String()}
}

func (p *zzProbe) genArrow() {
	full, err := SerializedArrowSchema()
	if err != nil {
		p.t.Fatal(err)
	}
	var cases []any
	add := func(name string, denied []string, file string, b []byte, inMemory *arrow.Schema) {
		rd, err := ipc.NewReader(bytes.NewReader(b))
		if err != nil {
			p.t.Fatalf("%s: decode: %v", name, err)
		}
		decoded := rd.Schema()
		if rd.Next() {
			p.t.Fatalf("%s: unexpected record batch", name)
		}
		rd.Release()
		desc := zzArrowSchema(decoded)
		if !bytes.Equal(zzMarshal(desc), zzMarshal(zzArrowSchema(inMemory))) {
			p.t.Fatalf("%s: decoded schema differs from the in-memory one", name)
		}
		sum := sha256.Sum256(b)
		p.writeFile("arrow/"+file, b)
		cases = append(cases, map[string]any{
			"name": name, "denied": denied, "file": file,
			"byte_len": len(b), "sha256": hex.EncodeToString(sum[:]),
			"schema": desc,
		})
	}
	add("serialized", nil, "events.arrows", full, ArrowSchema())
	for _, d := range zzDenylists {
		denied, err := parsePayloadColumnDenylist(d.denied)
		if err != nil {
			p.t.Fatal(err)
		}
		s := projectArrowSchema(denied)
		b, err := serializeArrowSchema(s)
		if err != nil {
			p.t.Fatal(err)
		}
		add("projected_"+d.name, d.denied, "projected_"+d.name+".arrows", b, s)
	}
	p.writeCases("arrow/schemas.json",
		"Arrow IPC stream bytes (schema message + end-of-stream marker; ipc.NewWriter + Close) in arrow/<file>: SerializedArrowSchema() (name serialized) and serializeArrowSchema(projectArrowSchema(parsePayloadColumnDenylist(denied))) (projected_*); schema = the decoded schema (ipc.NewReader), equal to the in-memory one. metadata = ordered [key, value] pairs.",
		cases)
}

// zzGenerator describes an input too large to store: kind "repeat" is an array of count copies
// of value, "repeat_then" appends the items of then, and "nested_maps"/"nested_lists" wrap the
// string "leaf" in depth single-key {"n": ...} maps or one-element arrays.
type zzGenerator struct {
	Kind  string `json:"kind"`
	Value any    `json:"value,omitempty"`
	Count int    `json:"count,omitempty"`
	Then  []any  `json:"then,omitempty"`
	Depth int    `json:"depth,omitempty"`
}

func (g zzGenerator) build() any {
	switch g.Kind {
	case "repeat", "repeat_then":
		items := make([]any, 0, g.Count+len(g.Then))
		for range g.Count {
			items = append(items, g.Value)
		}
		return append(items, g.Then...)
	case "nested_maps":
		var v any = "leaf"
		for range g.Depth {
			v = map[string]any{"n": v}
		}
		return v
	case "nested_lists":
		var v any = "leaf"
		for range g.Depth {
			v = []any{v}
		}
		return v
	}
	panic("unknown generator " + g.Kind)
}

func zzSmartTruncateResult(obj any, maxLen int, redact bool) map[string]any {
	b, truncated, err := smartTruncate(obj, maxLen, redact)
	m := zzOutput(b)
	m["truncated"] = truncated
	m["error"] = zzErrText(err)
	return m
}

func zzContent(t *testing.T, s string) *genai.Content {
	t.Helper()
	var c *genai.Content
	if err := json.Unmarshal([]byte(s), &c); err != nil {
		t.Fatalf("content %s: %v", s, err)
	}
	return c
}

func zzParts(parts []map[string]any) any {
	if parts == nil {
		return nil
	}
	return json.RawMessage(zzMarshal(parts))
}

func (p *zzProbe) genContent() {
	var parts, summaries []any
	for _, c := range zzContentCases {
		content := zzContent(p.t, c.content)
		withRedact, truncRedact := formatContentParts(content, c.maxLen, true)
		noRedact, truncNoRedact := formatContentParts(content, c.maxLen, false)
		public := FormatContentParts(content, c.maxLen)
		if !bytes.Equal(zzMarshal(public), zzMarshal(withRedact)) {
			p.t.Fatalf("%s: FormatContentParts differs from formatContentParts(..., true)", c.name)
		}
		parts = append(parts, map[string]any{
			"name": c.name, "content": zzRawJSON(p.t, c.content), "max_len": c.maxLen,
			"redact":    map[string]any{"parts": zzParts(withRedact), "truncated": truncRedact},
			"no_redact": map[string]any{"parts": zzParts(noRedact), "truncated": truncNoRedact},
		})
		sum := func(redact bool) map[string]any {
			s, st := contentSummary(content, c.maxLen, redact)
			r, rt := formatResponseContent(content, c.maxLen, redact)
			return map[string]any{"summary": s, "summary_truncated": st, "response": r, "response_truncated": rt}
		}
		summaries = append(summaries, map[string]any{
			"name": c.name, "content": zzRawJSON(p.t, c.content), "max_len": c.maxLen,
			"redact": sum(true), "no_redact": sum(false),
		})
	}
	p.writeCases("formatter/format_content_parts.json",
		"formatContentParts(content, max_len, redact) -> (parts, truncated); content is genai.Content JSON (null = nil *Content); parts null = Go nil slice (no parts). redact also equals FormatContentParts(content, max_len). part_index is an int; part_attributes is the exact SmartTruncate JSON text.",
		parts)
	p.writeCases("formatter/content_summaries.json",
		"contentSummary(content, max_len, redact) -> (summary, summary_truncated) and formatResponseContent(content, max_len, redact) -> (response, response_truncated).",
		summaries)
}

	var refs []any
	for _, r := range [][3]string{{"gs://b/o", "{}", ""}, {"gs://b/o", "{}", "us.conn"}, {"gs://b/o", `{"gcs_metadata":{"content_type":"text/plain"}}`, "c"}} {

// zzMineLiterals returns the distinct string literals (and constant concatenations of string
// literals) of path, except test names (name: fields and t.Run's first argument), messages of
// t.Error*/t.Fatal*/t.Log*/t.Skip*, struct tags and import paths, in sorted order.
func zzMineLiterals(t *testing.T, path string) []string {
	t.Helper()
	fset := token.NewFileSet()
	f, err := parser.ParseFile(fset, path, nil, parser.SkipObjectResolution)
	if err != nil {
		t.Fatal(err)
	}
	skip := map[ast.Expr]bool{}
	for _, imp := range f.Imports {
		skip[imp.Path] = true
	}
	ast.Inspect(f, func(n ast.Node) bool {
		switch x := n.(type) {
		case *ast.Field:
			if x.Tag != nil {
				skip[x.Tag] = true
			}
		case *ast.KeyValueExpr:
			if id, ok := x.Key.(*ast.Ident); ok && id.Name == "name" {
				skip[x.Value] = true
			}
		case *ast.CallExpr:
			if sel, ok := x.Fun.(*ast.SelectorExpr); ok {
				switch sel.Sel.Name {
				case "Run":
					if len(x.Args) > 0 {
						skip[x.Args[0]] = true
					}
				case "Errorf", "Error", "Fatalf", "Fatal", "Logf", "Log", "Skip", "Skipf", "Helper", "Sprintf":
					for _, a := range x.Args {
						skip[a] = true
					}
				}
			}
		}
		return true
	})
	var eval func(e ast.Expr) (string, bool)
	eval = func(e ast.Expr) (string, bool) {
		switch x := e.(type) {
		case *ast.BasicLit:
			if x.Kind != token.STRING {
				return "", false
			}
			s, err := strconv.Unquote(x.Value)
			return s, err == nil
		case *ast.ParenExpr:
			return eval(x.X)
		case *ast.BinaryExpr:
			if x.Op != token.ADD {
				return "", false
			}
			l, ok1 := eval(x.X)
			r, ok2 := eval(x.Y)
			return l + r, ok1 && ok2
		}
		return "", false
	}
	seen := map[string]bool{}
	ast.Inspect(f, func(n ast.Node) bool {
		e, ok := n.(ast.Expr)
		if !ok {
			return true
		}
		if skip[e] {
			return false
		}
		switch e.(type) {
		case *ast.BasicLit, *ast.BinaryExpr:
			if s, ok := eval(e); ok {
				seen[s] = true
			}
		}
		return true
	})
	out := make([]string, 0, len(seen))
	for s := range seen {
		out = append(out, s)
	}
	sort.Strings(out)
	return out
}

func zzSafe[T any](f func() T) (out T, panicked any) {
	defer func() {
		if r := recover(); r != nil {
			panicked = fmt.Sprint(r)
		}
	}()
	return f(), nil
}

// 6. REST wire captures.

// zzCanned is a response served by zzServer. Body may contain {{JOB_ID}}, replaced by the job ID
// of the last jobs.insert request before it is served.
type zzCanned struct {
	Status  int               `json:"status"`
	Headers map[string]string `json:"headers,omitempty"`
	Body    json.RawMessage   `json:"body"`
}

type zzRecordedRequest struct {
	Method   string              `json:"method"`
	Path     string              `json:"path"`
	RawQuery string              `json:"raw_query"`
	Query    map[string][]string `json:"query"`
	Headers  map[string][]string `json:"headers"`
	Body     json.RawMessage     `json:"body"`
	BodyText string              `json:"body_text"`
}

func (r zzRecordedRequest) BodyRaw() []byte { return []byte(r.BodyText) }

type zzExchange struct {
	Request  zzRecordedRequest `json:"request"`
	Response zzCanned          `json:"response"`
}

// zzExcludedHeaders are left out of recorded requests: credentials and client identification,
// which gadk sends its own way.
var zzExcludedHeaders = map[string]bool{"Authorization": true, "User-Agent": true, "X-Goog-Api-Client": true}

type zzServer struct {
	t         *testing.T
	mu        sync.Mutex
	srv       *httptest.Server
	canned    []zzCanned
	exchanges []zzExchange
	jobID     string
	jobIDs    map[string]bool
}

func newZZServer(t *testing.T, canned []zzCanned) *zzServer {
	s := &zzServer{t: t, canned: canned, jobIDs: map[string]bool{}}
	s.srv = httptest.NewServer(http.HandlerFunc(s.handle))
	return s
}

func (s *zzServer) close() { s.srv.Close() }

func (s *zzServer) client(ctx context.Context, project string) *bq.Client {
	c, err := bq.NewClient(ctx, project, option.WithEndpoint(s.srv.URL+"/bigquery/v2/"), option.WithoutAuthentication())
	if err != nil {
		s.t.Fatalf("bq.NewClient: %v", err)
	}
	return c
}

func (s *zzServer) handle(w http.ResponseWriter, r *http.Request) {
	body, _ := io.ReadAll(r.Body)
	s.mu.Lock()
	defer s.mu.Unlock()
	if r.Method == http.MethodPost && strings.HasSuffix(r.URL.Path, "/jobs") {
		var job struct {
			JobReference struct {
				JobID string `json:"jobId"`
			} `json:"jobReference"`
		}
		if json.Unmarshal(body, &job) == nil && job.JobReference.JobID != "" {
			s.jobID = job.JobReference.JobID
			s.jobIDs[s.jobID] = true
		}
	}
	norm := func(x string) string {
		if s.jobID == "" {
			return x
		}
		return strings.ReplaceAll(x, s.jobID, "{{JOB_ID}}")
	}
	headers := map[string][]string{}
	for k, v := range r.Header {
		if !zzExcludedHeaders[http.CanonicalHeaderKey(k)] {
			headers[http.CanonicalHeaderKey(k)] = v
		}
	}
	query := map[string][]string{}
	for k, v := range r.URL.Query() {
		nv := make([]string, len(v))
		for i, x := range v {
			nv[i] = norm(x)
		}
		query[k] = nv
	}
	req := zzRecordedRequest{
		Method: r.Method, Path: norm(r.URL.Path), RawQuery: norm(r.URL.RawQuery), Query: query, Headers: headers,
		BodyText: norm(string(body)),
	}
	if len(body) > 0 {
		var v any
		dec := json.NewDecoder(strings.NewReader(req.BodyText))
		dec.UseNumber()
		if err := dec.Decode(&v); err != nil {
			s.t.Errorf("request body is not JSON: %v", err)
		} else {
			req.Body = json.RawMessage(zzMarshal(v))
		}
	}
	var resp zzCanned
	if i := len(s.exchanges); i < len(s.canned) {
		resp = s.canned[i]
	} else {
		resp = zzCanned{Status: 599, Body: json.RawMessage(`{"error":{"code":599,"message":"probe: unexpected request"}}`)}
		s.t.Errorf("unexpected request #%d: %s %s", i, r.Method, r.URL)
	}
	s.exchanges = append(s.exchanges, zzExchange{Request: req, Response: resp})
	for k, v := range resp.Headers {
		w.Header().Set(k, v)
	}
	if w.Header().Get("Content-Type") == "" {
		w.Header().Set("Content-Type", "application/json; charset=UTF-8")
	}
	w.WriteHeader(resp.Status)
	_, _ = w.Write([]byte(strings.ReplaceAll(string(resp.Body), "{{JOB_ID}}", s.jobID)))
}

func zzOK(body string) zzCanned { return zzCanned{Status: 200, Body: json.RawMessage(body)} }

// zzGoogleError is a Google API JSON error response.
func zzGoogleError(code int, status, reason, message string) zzCanned {
	b := zzMarshal(map[string]any{"error": map[string]any{
		"code": code, "message": message, "status": status,
		"errors": []any{map[string]any{"message": message, "domain": "global", "reason": reason}},
	}})
	return zzCanned{Status: code, Body: json.RawMessage(b)}
}

func (p *zzProbe) zzTableResource(schema bq.Schema, etag string, labels map[string]string) string {
	m := map[string]any{
		"kind": "bigquery#table", "etag": etag, "id": "test-project:agent_analytics.agent_events",
		"tableReference": map[string]any{"projectId": "test-project", "datasetId": "agent_analytics", "tableId": "agent_events"},
		"schema":         p.zzRESTSchema(schema),
		"type":           "TABLE",
	}
	if labels != nil {
		m["labels"] = labels
	}
	return string(zzMarshal(m))
}

// zzSchemaWithout returns EventsSchema without the top-level fields drop and without
// content_parts.object_ref.details when dropDetails is set.
func zzSchemaWithout(dropDetails bool, drop ...string) bq.Schema {
	var out bq.Schema
	for _, f := range EventsSchema() {
		if zzContains(drop, f.Name) {
			continue
		}
		if dropDetails && f.Name == "content_parts" {
			cp := *f
			cp.Schema = nil
			for _, sf := range f.Schema {
				sfc := *sf
				if sf.Name == "object_ref" {
					sfc.Schema = nil
					for _, ssf := range sf.Schema {
						if ssf.Name != "details" {
							sfc.Schema = append(sfc.Schema, ssf)
						}
					}
				}
				cp.Schema = append(cp.Schema, &sfc)
			}
			out = append(out, &cp)
			continue
		}
		out = append(out, f)
	}
	return out
}

func zzContains(list []string, s string) bool {
	for _, x := range list {
		if x == s {
			return true
		}
	}
	return false
}

// zzJobIDRe is the shape of the bigquery client's random job IDs (hex of 16 random bytes).
var zzJobIDRe = regexp.MustCompile(`^[0-9a-f]{32}$`)

// zzErrType names the error's dynamic type chain for documentation.
func zzErrType(err error) any {
	if err == nil {
		return nil
	}
	var types []string
	for e := err; e != nil; e = errors.Unwrap(e) {
		types = append(types, fmt.Sprintf("%T", e))
	}
	return types
}
