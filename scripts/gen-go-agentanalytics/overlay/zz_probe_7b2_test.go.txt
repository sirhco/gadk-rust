func zzSHA256(b []byte) string {
	sum := sha256.Sum256(b)
	return hex.EncodeToString(sum[:])
}

// zzNormJSON decodes JSON text (numbers kept exact) and re-encodes it with sorted keys. It is
// how protojson output is written: protojson randomizes its whitespace between builds.
func zzNormJSON(t *testing.T, b []byte) json.RawMessage {
	t.Helper()
	dec := json.NewDecoder(bytes.NewReader(b))
	dec.UseNumber()
	var v any
	if err := dec.Decode(&v); err != nil {
		t.Fatalf("zzNormJSON: %v", err)
	}
	return json.RawMessage(zzMarshal(v))
}

func zzProtoJSON(t *testing.T, m proto.Message) json.RawMessage {
	t.Helper()
	b, err := protojson.Marshal(m)
	if err != nil {
		t.Fatalf("protojson: %v", err)
	}
	return zzNormJSON(t, b)
}

// zzDecodeSpec parses JSON text with json.Number numbers.
func zzDecodeSpec(t *testing.T, s string) any {
	t.Helper()
	dec := json.NewDecoder(strings.NewReader(s))
	dec.UseNumber()
	var v any
	if err := dec.Decode(&v); err != nil {
		t.Fatalf("zzDecodeSpec(%q): %v", s, err)
	}
	return v
}

// zzBuildValue builds the Go row value that a typed value spec {"t": tag, "v": value} stands
// for (SCENARIO.md "Row specs").
func zzBuildValue(t *testing.T, spec any) any {
	t.Helper()
	m, ok := spec.(map[string]any)
	if !ok {
		t.Fatalf("value spec %v is not an object", spec)
	}
	tag, _ := m["t"].(string)
	v := m["v"]
	num := func() json.Number {
		n, ok := v.(json.Number)
		if !ok {
			t.Fatalf("value spec %v: v is not a number", spec)
		}
		return n
	}
	switch tag {
	case "null":
		return nil
	case "str":
		return v.(string)
	case "ts":
		ts, err := time.Parse(time.RFC3339Nano, v.(string))
		if err != nil {
			t.Fatalf("value spec %v: %v", spec, err)
		}
		return ts
	case "bool":
		return v.(bool)
	case "int64":
		n, err := num().Int64()
		if err != nil {
			t.Fatal(err)
		}
		return n
	case "int":
		n, err := num().Int64()
		if err != nil {
			t.Fatal(err)
		}
		return int(n)
	case "int32":
		n, err := num().Int64()
		if err != nil {
			t.Fatal(err)
		}
		return int32(n)
	case "float64":
		f, err := num().Float64()
		if err != nil {
			t.Fatal(err)
		}
		return f
	case "parts":
		list := v.([]any)
		out := make([]map[string]any, len(list))
		for i, e := range list {
			if e == nil {
				continue // a nil map
			}
			out[i] = zzBuildRow(t, e)
		}
		return out
	case "map":
		return zzBuildRow(t, v)
	case "any_list":
		list := v.([]any)
		out := make([]any, len(list))
		for i, e := range list {
			out[i] = zzBuildValue(t, e)
		}
		return out
	case "str_map":
		out := map[string]string{}
		for k, x := range v.(map[string]any) {
			out[k] = x.(string)
		}
		return out
	case "bytes":
		b, err := base64.StdEncoding.DecodeString(v.(string))
		if err != nil {
			t.Fatal(err)
		}
		return b
	}
	t.Fatalf("value spec %v: unknown tag %q", spec, tag)
	return nil
}

// zzBuildRow builds a row map from a row spec (an object of value specs, or null for a nil map).
func zzBuildRow(t *testing.T, spec any) map[string]any {
	t.Helper()
	if spec == nil {
		return nil
	}
	m, ok := spec.(map[string]any)
	if !ok {
		t.Fatalf("row spec %v is not an object", spec)
	}
	out := make(map[string]any, len(m))
	for k, s := range m {
		out[k] = zzBuildValue(t, s)
	}
	return out
}

func zzBuildRows(t *testing.T, specs []any) []map[string]any {
	t.Helper()
	rows := make([]map[string]any, len(specs))
	for i, s := range specs {
		rows[i] = zzBuildRow(t, s)
	}
	return rows
}

func zzB(b bool) map[string]any    { return map[string]any{"t": "bool", "v": b} }
func zzI64(n int64) map[string]any {
	return map[string]any{"t": "int64", "v": json.Number(fmt.Sprint(n))}
}
func zzInt(n int) map[string]any { return map[string]any{"t": "int", "v": json.Number(fmt.Sprint(n))} }
func zzI32(n int32) map[string]any {
	return map[string]any{"t": "int32", "v": json.Number(fmt.Sprint(n))}
}
func zzF64(lit string) map[string]any { return map[string]any{"t": "float64", "v": json.Number(lit)} }
func zzNull() map[string]any          { return map[string]any{"t": "null"} }
func zzPartsV(parts ...any) map[string]any {
	return map[string]any{"t": "parts", "v": append([]any{}, parts...)}
}
func zzMap(m map[string]any) map[string]any { return map[string]any{"t": "map", "v": m} }
func zzAnyList(items ...any) map[string]any {
	return map[string]any{"t": "any_list", "v": append([]any{}, items...)}
}
func zzStrMap(m map[string]any) map[string]any { return map[string]any{"t": "str_map", "v": m} }
func zzBytes(b []byte) map[string]any {
	return map[string]any{"t": "bytes", "v": base64.StdEncoding.EncodeToString(b)}
}

// zzArrowValue describes element i of arr as JSON: null, a string, a bool, an int64 (timestamps
// as microseconds since the epoch), a list, or a struct as an object of all its fields.
func zzArrowValue(arr arrow.Array, i int) any {
	if arr.IsNull(i) {
		return nil
	}
	switch a := arr.(type) {
	case *array.Timestamp:
		return int64(a.Value(i))
	case *array.String:
		return a.Value(i)
	case *array.Int64:
		return a.Value(i)
	case *array.Boolean:
		return a.Value(i)
	case *array.List:
		start, end := a.ValueOffsets(i)
		out := []any{}
		for j := start; j < end; j++ {
			out = append(out, zzArrowValue(a.ListValues(), int(j)))
		}
		return out
	case *array.Struct:
		st := a.DataType().(*arrow.StructType)
		out := map[string]any{}
		for k, f := range st.Fields() {
			out[f.Name] = zzArrowValue(a.Field(k), i)
		}
		return out
	}
	panic("zzArrowValue: unexpected array " + arr.DataType().String())
}

func zzDecodeRecord(rec arrow.RecordBatch) []any {
	rows := []any{}
	for i := 0; i < int(rec.NumRows()); i++ {
		row := map[string]any{}
		for c, f := range rec.Schema().Fields() {
			row[f.Name] = zzArrowValue(rec.Column(c), i)
		}
		rows = append(rows, row)
	}
	return rows
}

// zzReadBatch decodes Go's serializeRecord bytes: the schema stream of schemaBytes without its
// 8-byte end-of-stream marker, then the record batch message, then the marker.
func zzReadBatch(t *testing.T, schemaBytes, batchBytes []byte) arrow.RecordBatch {
	t.Helper()
	eos := []byte{0xff, 0xff, 0xff, 0xff, 0, 0, 0, 0}
	if !bytes.HasSuffix(schemaBytes, eos) {
		t.Fatal("schema bytes do not end with the end-of-stream marker")
	}
	var stream []byte
	stream = append(stream, schemaBytes[:len(schemaBytes)-len(eos)]...)
	stream = append(stream, batchBytes...)
	stream = append(stream, eos...)
	rd, err := ipc.NewReader(bytes.NewReader(stream))
	if err != nil {
		t.Fatalf("ipc.NewReader: %v", err)
	}
	if !rd.Next() {
		t.Fatalf("no record batch: %v", rd.Err())
	}
	rec := rd.RecordBatch()
	rec.Retain()
	if rd.Next() {
		t.Fatal("more than one record batch")
	}
	rd.Release()
	return rec
}

func (p *zzProbe) genRecordBatches() {
	var out []any
	for _, bc := range zzBatchCases() {
		var deniedList []string
		for _, d := range zzDenylists {
			if d.name == bc.denied {
				deniedList = d.denied
			}
		}
		rowsSpec := zzRoundTrip(p.t, bc.rows).([]any)
		bp := zzNewProcessor(p.t, Config{PayloadColumnDenylist: deniedList})
		rows := zzBuildRows(p.t, rowsSpec)
		rec := bp.buildRecordBatch(rows)
		b := serializeRecord(rec, nil)
		if b == nil {
			p.t.Fatalf("%s: serializeRecord failed", bc.name)
		}
		schemaBytes, err := serializeArrowSchema(bp.arrowSchema)
		if err != nil {
			p.t.Fatal(err)
		}
		decoded := zzReadBatch(p.t, schemaBytes, b)
		if !array.RecordEqual(decoded, rec) {
			p.t.Fatalf("%s: decoded record differs from the built one", bc.name)
		}
		file := bc.name + ".batch"
		p.writeFile("record_batch/"+file, b)
		var nullCounts [][2]any
		for c, f := range rec.Schema().Fields() {
			nullCounts = append(nullCounts, [2]any{f.Name, rec.Column(c).NullN()})
		}
		out = append(out, map[string]any{
			"name": bc.name, "about": bc.about, "denied": deniedList, "schema_file": zzSchemaFile(bc.denied),
			"rows": rowsSpec, "file": file, "byte_len": len(b), "sha256": zzSHA256(b),
			"num_rows": rec.NumRows(), "null_counts": nullCounts, "decoded": zzDecodeRecord(decoded),
		})
		decoded.Release()
		rec.Release()
		bp.Close()
	}
	p.writeCases("record_batch/corpus.json",
		"bp.buildRecordBatch(rows) then serializeRecord(record, nil) on a BatchProcessor built with PayloadColumnDenylist = denied (its arrowSchema = the schema in schema_file). file holds the record batch message bytes (ipc.GetRecordBatchPayload + WritePayload: no schema, no end-of-stream marker); decode them by appending them to schema_file's bytes minus their last 8 (end-of-stream) bytes and then the marker. rows are row specs (SCENARIO.md); decoded = the decoded rows, one object per row with every column of the schema (timestamps as int64 microseconds, lists as arrays, structs as objects with all their fields, NULL as null); null_counts = [column, null count] in schema order.",
		out)

	p.genEstimates()
}

func (p *zzProbe) genAppendRows() {
	batchCase := func(name string) ([]byte, int64) {
		for _, bc := range zzBatchCases() {
			if bc.name != name {
				continue
			}
			var deniedList []string
			for _, d := range zzDenylists {
				if d.name == bc.denied {
					deniedList = d.denied
				}
			}
			bp := zzNewProcessor(p.t, Config{PayloadColumnDenylist: deniedList})
			defer bp.Close()
			rec := bp.buildRecordBatch(zzBuildRows(p.t, zzRoundTrip(p.t, bc.rows).([]any)))
			defer rec.Release()
			return serializeRecord(rec, nil), rec.NumRows()
		}
		p.t.Fatalf("no batch case %s", name)
		return nil, 0
	}
	type reqCase struct {
		name        string
		about       string
		stream      string
		exactlyOnce bool
		offset      int64
		denied      string
		batch       string // record_batch case, "" for nil bytes
	}
	named := "projects/test-project/datasets/agent_analytics/tables/agent_events/streams/Cizc0123"
	def := "projects/test-project/datasets/agent_analytics/tables/agent_events/streams/_default"
	cases := []reqCase{
		{"default_stream", "the _default stream, one row", def, false, 0, "none", "all_columns"},
		{"named_stream", "a committed stream name", named, false, 0, "none", "all_columns"},
		{"empty_stream_name", "streamName \"\": write_stream is empty and omitted from the wire", "", false, 0, "none", "all_columns"},
		{"multi_row", "three rows", def, false, 0, "none", "multi_row"},
		{"empty_batch", "a zero-row record batch", def, false, 0, "none", "empty_batch"},
		{"nil_batch_bytes", "batchBytes nil and numRows 0: rows message present with empty bytes and row_count 0 (both default, so arrow_rows.rows encodes as an empty message)", def, false, 0, "none", ""},
		{"projected_schema", "the content/content_parts/attributes/latency_ms denylist: the writer schema is the projected one", def, false, 0, "all", "projected_all"},
		{"exactly_once_offset_0", "ExactlyOnceDelivery with streamOffset 0: offset is set (Int64Value 0, an empty wrapper message)", named, true, 0, "none", "all_columns"},
		{"exactly_once_offset_42", "ExactlyOnceDelivery with streamOffset 42", named, true, 42, "none", "multi_row"},
		{"exactly_once_offset_large", "ExactlyOnceDelivery with streamOffset 2^40+3", named, true, 1<<40 + 3, "none", "all_columns"},
	}
	var out []any
	for _, c := range cases {
		var deniedList []string
		for _, d := range zzDenylists {
			if d.name == c.denied {
				deniedList = d.denied
			}
		}
		schemaBytes, err := serializeArrowSchema(projectArrowSchema(zzDeniedSet(p.t, deniedList)))
		if err != nil {
			p.t.Fatal(err)
		}
		var batchBytes []byte
		var numRows int64
		batchFile := any(nil)
		if c.batch != "" {
			batchBytes, numRows = batchCase(c.batch)
			batchFile = "record_batch/" + c.batch + ".batch"
		}
		sw := newStreamWriter(context.Background(), nil, c.stream, schemaBytes, Config{ExactlyOnceDelivery: c.exactlyOnce})
		sw.streamOffset.Store(c.offset)
		sw.mu.Lock()
		req := sw.newAppendRowsRequestLocked(batchBytes, numRows)
		sw.mu.Unlock()
		b, err := proto.MarshalOptions{Deterministic: true}.Marshal(req)
		if err != nil {
			p.t.Fatal(err)
		}
		plain, err := proto.Marshal(req)
		if err != nil || !bytes.Equal(plain, b) {
			p.t.Fatalf("%s: proto.Marshal differs from the deterministic encoding", c.name)
		}
		size := sw.serializedRequestSize(batchBytes, numRows)
		if size != len(b) {
			p.t.Fatalf("%s: serializedRequestSize %d != len %d", c.name, size, len(b))
		}
		var offset any
		if req.Offset != nil {
			offset = req.Offset.GetValue()
		}
		file := c.name + ".pb"
		p.writeFile("append_rows/"+file, b)
		out = append(out, map[string]any{
			"name": c.name, "about": c.about,
			"stream_name": c.stream, "exactly_once": c.exactlyOnce, "stream_offset": c.offset,
			"denied": deniedList, "writer_schema_file": zzSchemaFile(c.denied), "batch_file": batchFile, "num_rows": numRows,
			"trace_id": req.TraceId, "offset": offset,
			"file": file, "byte_len": len(b), "sha256": zzSHA256(b),
			"serialized_request_size": size,
			"protojson":               zzProtoJSON(p.t, req),
		})
	}
	p.writeCases("append_rows/requests.json",
		"streamWriter.newAppendRowsRequestLocked(batchBytes, numRows) on newStreamWriter(ctx, nil, stream_name, schemaBytes, Config{ExactlyOnceDelivery: exactly_once}) with streamOffset = stream_offset; schemaBytes = writer_schema_file's bytes, batchBytes = batch_file's bytes (null = nil). file = proto.Marshal(req) (equal to the deterministic encoding); serialized_request_size = serializedRequestSize(...) = proto.Size(req) = byte_len. trace_id is streamWriterTraceID() of the probe binary (see trace_ids.json): substitute it when comparing a request built with another trace id. protojson = protojson.Marshal(req) re-encoded with sorted keys (bytes as std base64, int64 as strings).",
		out)

	var tids []any
	for _, v := range []string{"", "1.2.3", "2.0.0", "v1.0.0", "1.2.3-rc.1+build.5", "(devel)", "1.0.0 beta"} {
		tids = append(tids, map[string]any{"version": v, "trace_id": streamWriterTraceIDForVersion(v)})
	}
	p.writeJSON("append_rows/trace_ids.json", map[string]any{
		"about":             "streamWriterTraceIDForVersion(version) cases; probe_adk_version = adkVersion() and probe_trace_id = streamWriterTraceID() of the probe's test binary (the trace_id of requests.json).",

// zzWriteServer is an in-process BigQuery Write server that records the incoming metadata of
// every call and answers the scripted error codes (one per call, in order) before succeeding.
type zzWriteServer struct {
	storagepb.UnimplementedBigQueryWriteServer
	t      *testing.T
	mu     sync.Mutex
	calls  []map[string]any
	script map[string][]codes.Code
	counts map[string]int
}

func (s *zzWriteServer) begin(ctx context.Context, rpc string, req proto.Message) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	md, _ := metadata.FromIncomingContext(ctx)
	call := map[string]any{"rpc": rpc, "metadata": zzMetadata(md)}
	if dl, ok := ctx.Deadline(); ok {
		call["timeout_s"] = int64(math.Round(time.Until(dl).Seconds()))
	} else {
		call["timeout_s"] = nil
	}
	if req != nil {
		call["request"] = zzProtoJSON(s.t, req)
	}
	s.calls = append(s.calls, call)
	n := s.counts[rpc]
	s.counts[rpc] = n + 1
	if sc := s.script[rpc]; n < len(sc) && sc[n] != codes.OK {
		return status.Errorf(sc[n], "zz scripted %s", sc[n])
	}
	return nil
}

func (s *zzWriteServer) CreateWriteStream(ctx context.Context, req *storagepb.CreateWriteStreamRequest) (*storagepb.WriteStream, error) {
	if err := s.begin(ctx, "CreateWriteStream", req); err != nil {
		return nil, err
	}
	return &storagepb.WriteStream{Name: req.GetParent() + "/streams/zz-stream-1", Type: storagepb.WriteStream_COMMITTED}, nil
}

func (s *zzWriteServer) FinalizeWriteStream(ctx context.Context, req *storagepb.FinalizeWriteStreamRequest) (*storagepb.FinalizeWriteStreamResponse, error) {
	if err := s.begin(ctx, "FinalizeWriteStream", req); err != nil {
		return nil, err
	}
	return &storagepb.FinalizeWriteStreamResponse{RowCount: 3}, nil
}

func (s *zzWriteServer) AppendRows(stream storagepb.BigQueryWrite_AppendRowsServer) error {
	if err := s.begin(stream.Context(), "AppendRows", nil); err != nil {
		return err
	}
	for {
		req, err := stream.Recv()
		if err != nil {
			return nil
		}
		s.mu.Lock()
		var offset any
		if req.Offset != nil {
			offset = req.Offset.GetValue()
		}
		call := s.calls[len(s.calls)-1]
		reqs, _ := call["requests"].([]any)
		call["requests"] = append(reqs, map[string]any{
			"write_stream": req.GetWriteStream(), "trace_id": req.GetTraceId(), "offset": offset,
			"row_count": req.GetArrowRows().GetRows().GetRowCount(),
		})
		s.mu.Unlock()
		res := &storagepb.AppendRowsResponse{
			Response:    &storagepb.AppendRowsResponse_AppendResult_{AppendResult: &storagepb.AppendRowsResponse_AppendResult{Offset: req.Offset}},
			WriteStream: req.GetWriteStream(),
		}
		if err := stream.Send(res); err != nil {
			return nil
		}
	}
}

// zzMetadata returns the incoming metadata with sorted keys, leaving out :authority (the
// listener's address) and with the Go version in x-goog-api-client templated.
func zzMetadata(md metadata.MD) map[string][]string {
	out := map[string][]string{}
	for k, vs := range md {
		if k == ":authority" {
			continue
		}
		nv := make([]string, len(vs))
		for i, v := range vs {
			nv[i] = strings.ReplaceAll(v, "gl-go/"+gax.GoVersion, "gl-go/{{GO_VERSION}}")
		}
		out[k] = nv
	}
	return out
}

// zzCallSettings resolves gax call options and reads the timeout and the retryer's codes and
// backoff (unexported fields, read by reflection).
func zzCallSettings(opts []gax.CallOption) map[string]any {
	var s gax.CallSettings
	for _, o := range opts {
		o.Resolve(&s)
	}
	out := map[string]any{"timeout_ns": reflect.ValueOf(s).FieldByName("timeout").Int()}
	if s.Retry == nil {
		out["retry"] = nil
		return out
	}
	r := reflect.ValueOf(s.Retry()).Elem()
	var cs []string
	codesV := r.FieldByName("codes")
	for i := 0; i < codesV.Len(); i++ {
		cs = append(cs, codes.Code(codesV.Index(i).Uint()).String())
	}
	bo := r.FieldByName("backoff")
	out["retry"] = map[string]any{
		"codes":      cs,
		"initial_ns": bo.FieldByName("Initial").Int(),
		"max_ns":     bo.FieldByName("Max").Int(),
		"multiplier": bo.FieldByName("Multiplier").Float(),
	}
	return out
}

// zzRetryCodes returns the retry codes of the default call options of an RPC.
func zzRetryCodes(opts []gax.CallOption) []codes.Code {
	var s gax.CallSettings
	for _, o := range opts {
		o.Resolve(&s)
	}
	if s.Retry == nil {
		return nil
	}
	codesV := reflect.ValueOf(s.Retry()).Elem().FieldByName("codes")
	var cs []codes.Code
	for i := 0; i < codesV.Len(); i++ {
		cs = append(cs, codes.Code(codesV.Index(i).Uint()))
	}
	return cs
}

func zzStatusOf(err error) any {
	if err == nil {
		return nil
	}
	st, ok := status.FromError(err)
	if !ok {
		return map[string]any{"code": nil, "error": err.Error()}
	}
	return map[string]any{"code": st.Code().String(), "message": st.Message(), "error": err.Error()}
}

func (p *zzProbe) genGapicRetries() {
	g0 := p.newZZGapic(nil)
	defaults := g0.client.CallOptions
	g0.close()
	fast := func(opts []gax.CallOption, timeout time.Duration) []gax.CallOption {
		cs := zzRetryCodes(opts)
		out := []gax.CallOption{gax.WithRetry(func() gax.Retryer {
			return gax.OnCodes(cs, gax.Backoff{Initial: time.Millisecond, Max: time.Millisecond, Multiplier: 1.3})
		})}
		if timeout > 0 {
			out = append([]gax.CallOption{gax.WithTimeout(timeout)}, out...)
		}
		return out
	}
	scripted := []codes.Code{codes.Unavailable, codes.DeadlineExceeded, codes.ResourceExhausted, codes.Internal, codes.Aborted, codes.Unknown, codes.PermissionDenied, codes.NotFound, codes.InvalidArgument}
	cfg := Config{ProjectID: "test-project", DatasetID: "agent_analytics", TableName: "agent_events"}
	var out []any
	for _, rpc := range []string{"CreateWriteStream", "FinalizeWriteStream"} {
		for _, code := range scripted {
			script := []codes.Code{code, code, codes.OK}
			g := p.newZZGapic(map[string][]codes.Code{rpc: script})
			g.client.CallOptions.CreateWriteStream = fast(defaults.CreateWriteStream, 1200*time.Second)
			g.client.CallOptions.FinalizeWriteStream = fast(defaults.FinalizeWriteStream, 600*time.Second)
			var err error
			switch rpc {
			case "CreateWriteStream":
				_, err = createCommittedWriteStream(context.Background(), g.client, cfg)
			case "FinalizeWriteStream":
				_, err = g.client.FinalizeWriteStream(context.Background(), &storagepb.FinalizeWriteStreamRequest{Name: "projects/p/datasets/d/tables/t/streams/s"})
			}
			attempts := len(g.calls())
			g.close()
			var sc []string
			for _, c := range script {
				sc = append(sc, c.String())
			}
			out = append(out, map[string]any{"rpc": rpc, "script": sc, "attempts": attempts, "status": zzStatusOf(err)})
		}
	}
	// AppendRows: the server answers the scripted status at the start of the stream. grpc-go's
	// NewStream does not wait for the server, so opening succeeds and gax's open retry (on
	// Unavailable) never sees the status: it surfaces on Recv, and Send then reports io.EOF.
	for _, code := range []codes.Code{codes.Unavailable, codes.ResourceExhausted, codes.Internal} {
		g := p.newZZGapic(map[string][]codes.Code{"AppendRows": {code}})
		g.client.CallOptions.AppendRows = fast(defaults.AppendRows, 0)
		ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
		stream, openErr := g.client.AppendRows(ctx)
		var sendErr, recvErr error
		if openErr == nil {
			// Recv first, so that the status has arrived and Send deterministically sees
			// the ended stream.
			_, recvErr = stream.Recv()
			sendErr = stream.Send(&storagepb.AppendRowsRequest{WriteStream: "projects/p/datasets/d/tables/t/streams/_default"})
		}
		cancel()
		attempts := len(g.calls())
		g.close()
		sendDesc := any(nil)
		if sendErr != nil {
			sendDesc = map[string]any{"is_eof": errors.Is(sendErr, io.EOF), "error": sendErr.Error()}
		}
		out = append(out, map[string]any{
			"rpc": "AppendRows", "script": []string{code.String()}, "attempts": attempts,
			"open_status": zzStatusOf(openErr), "send": sendDesc, "status": zzStatusOf(recvErr),
		})
	}
	p.writeCases("gapic/retry.json",
		"Retries of the gapic client against scripted answers (script = the status of each call in order; OK = success). The probe keeps the default call options' retry codes and timeouts (call_options.json) but shortens the backoff to 1ms, since gax's pauses are random: only attempts (calls the server saw) and the final status are recorded. CreateWriteStream goes through createCommittedWriteStream, FinalizeWriteStream through the client. AppendRows: the server answers the scripted status as soon as the stream starts; open_status = the error of client.AppendRows(ctx) (grpc-go's NewStream does not wait for the server, so it is nil and gax's open retry never sees the status), status = the first Recv's error, send = the error of a Send made after that Recv (io.EOF on an ended stream; streamWriter.append drains Recv for the status when Send reports io.EOF).",
		out)
}

type zzGCSReply struct {
	status  int
	headers map[string]string

type zzGCSServer struct {
	t         *testing.T
	mu        sync.Mutex
	srv       *httptest.Server
	reply     func(i int, r *http.Request, body []byte) zzGCSReply
	exchanges []any
	tokens    map[string]string
	invIDs    map[string]string
}

// zzGCSTemplate replaces the server address, the gl-go version and recorded random tokens.
func (s *zzGCSServer) template(v string) string {
	v = strings.ReplaceAll(v, s.srv.URL, "{{SERVER}}")
	v = strings.ReplaceAll(v, "gl-go/"+gax.GoVersion, "gl-go/{{GO_VERSION}}")
	for tok, name := range s.tokens {
		v = strings.ReplaceAll(v, tok, name)
	}
	for id, name := range s.invIDs {
		v = strings.ReplaceAll(v, id, name)
	}
	return v
}

func (s *zzGCSServer) learnTokens(r *http.Request) {
	if tok := r.Header.Get("X-Goog-Gcs-Idempotency-Token"); tok != "" {
		if _, ok := s.tokens[tok]; !ok {
			s.tokens[tok] = fmt.Sprintf("{{IDEMPOTENCY_TOKEN_%d}}", len(s.tokens)+1)
		}
	}
	for _, field := range strings.Fields(r.Header.Get("X-Goog-Api-Client")) {
		if id, ok := strings.CutPrefix(field, "gccl-invocation-id/"); ok {
			if _, seen := s.invIDs[id]; !seen {
				s.invIDs[id] = fmt.Sprintf("{{INVOCATION_ID_%d}}", len(s.invIDs)+1)
			}
		}
	}
}

func (s *zzGCSServer) describeBody(r *http.Request, body []byte) map[string]any {
	d := map[string]any{"len": len(body)}
	mediaType, params, err := mime.ParseMediaType(r.Header.Get("Content-Type"))
	if err == nil && strings.HasPrefix(mediaType, "multipart/") {
		mr := multipart.NewReader(bytes.NewReader(body), params["boundary"])
		var parts []any
		for {
			part, err := mr.NextPart()
			if err == io.EOF {
				break
			}
			if err != nil {
				s.t.Fatalf("multipart: %v", err)
			}
			data, err := io.ReadAll(part)
			if err != nil {
				s.t.Fatal(err)
			}
			hdr := map[string][]string{}
			for k, v := range part.Header {
				hdr[k] = v
			}
			pd := map[string]any{"headers": hdr, "len": len(data), "sha256": zzSHA256(data)}
			if strings.HasPrefix(part.Header.Get("Content-Type"), "application/json") {
				pd["text"] = string(data)
				pd["json"] = zzNormJSON(s.t, data)
			}
			parts = append(parts, pd)
		}
		d["multipart"] = parts
		d["boundary_len"] = len(params["boundary"])
		return d
	}
	d["sha256"] = zzSHA256(body)
	if len(body) > 0 && len(body) < 64*1024 && json.Valid(body) {
		d["text"] = s.template(string(body))
		d["json"] = zzNormJSON(s.t, []byte(s.template(string(body))))
	}
	return d
}

func (s *zzGCSServer) handle(w http.ResponseWriter, r *http.Request) {
	body, _ := io.ReadAll(r.Body)
	s.mu.Lock()
	defer s.mu.Unlock()
	s.learnTokens(r)
	headers := map[string][]string{}
	for k, vs := range r.Header {
		ck := http.CanonicalHeaderKey(k)
		if ck == "User-Agent" || ck == "Authorization" {
			continue
		}
		nv := make([]string, len(vs))
		for i, v := range vs {
			if ck == "Content-Type" {
				if _, params, err := mime.ParseMediaType(v); err == nil && params["boundary"] != "" {
					v = strings.ReplaceAll(v, params["boundary"], "{{BOUNDARY}}")
				}
			}
			nv[i] = s.template(v)
		}
		headers[ck] = nv
	}
	query := map[string][]string{}
	for k, vs := range r.URL.Query() {
		nv := make([]string, len(vs))
		for i, v := range vs {
			nv[i] = s.template(v)
		}
		query[k] = nv
	}
	rep := s.reply(len(s.exchanges), r, body)
	repHeaders := map[string]string{}
	for k, v := range rep.headers {
		repHeaders[k] = s.template(v)
	}
	s.exchanges = append(s.exchanges, map[string]any{
		"request": map[string]any{
			"method": r.Method, "path": r.URL.Path, "raw_query": s.template(r.URL.RawQuery), "query": query,
			"headers": headers, "body": s.describeBody(r, body),
		},
		"response": map[string]any{"status": rep.status, "headers": repHeaders, "body": s.template(rep.body)},
	})
	for k, v := range rep.headers {
		w.Header().Set(k, v)
	}
	if w.Header().Get("Content-Type") == "" {
		w.Header().Set("Content-Type", "application/json; charset=UTF-8")
	}
	w.WriteHeader(rep.status)
	_, _ = w.Write([]byte(rep.body))
}

// zzPattern returns n bytes with byte i = i % 251.
func zzPattern(n int) []byte {
	b := make([]byte, n)
	for i := range b {
		b[i] = byte(i % 251)
	}

// zzResumableReply answers a resumable upload as Cloud Storage does for a client that sends
// X-GUploader-No-308: yes. The initiation request (no upload_id) gets 200 and a Location; a
// chunk whose Content-Range total is "*" gets 200 with X-Http-Status-Code-Override: 308 and the
// Range received so far; the final chunk gets the object. failChunk[n] = status answers the
// n-th chunk request (1-based, counting resends) with that error instead.
func zzResumableReply(srvURL func() string, name string, contentType string, failChunk map[int]int) func(i int, r *http.Request, body []byte) zzGCSReply {
	received := 0
	chunk := 0
	return func(i int, r *http.Request, body []byte) zzGCSReply {
		if r.URL.Query().Get("upload_id") == "" {
			return zzGCSReply{status: 200, headers: map[string]string{
				"Location": srvURL() + "/upload/storage/v1/b/test-bucket/o?uploadType=resumable&upload_id=zz-upload-1",
			}, body: ""}
		}
		chunk++
		if code, ok := failChunk[chunk]; ok {
			return zzStorageError(code, "backendError")
		}
		cr := r.Header.Get("Content-Range")
		received += len(body)
		if strings.HasSuffix(cr, "/*") {
			return zzGCSReply{status: 200, headers: map[string]string{
				"X-Http-Status-Code-Override": "308",
				"Range":                       fmt.Sprintf("bytes=0-%d", received-1),
			}}
		}
		return zzObjectReply(name, received, contentType)
	}
}

		}
	}
	resumable := func(name, ct string, fail map[int]int) func(s *zzGCSServer) func(int, *http.Request, []byte) zzGCSReply {
		return func(s *zzGCSServer) func(int, *http.Request, []byte) zzGCSReply {
			return zzResumableReply(func() string { return s.srv.URL }, name, ct, fail)
		}
	}
	textPath := "logs/2026-03-04/trace/span_evt_c0_p1.txt"
	pngPath := "logs/2026-03-04/t/s_e_c0_p0.png"
	binPath := "big/2026-03-04/t/s_e_c0_p0.bin"
	text := []byte("héllo, offloaded text ✓")
	cases := []gcsCase{
		{"multipart_small", "a 7-byte image/png payload: one multipart upload (ChunkSize = len+1)", pngPath, "image/png", small,
			map[string]any{"base64": base64.StdEncoding.EncodeToString(small)}, okMultipart(pngPath, "image/png", len(small))},
		{"multipart_text", "a UTF-8 text payload with the text upload content type", textPath, textUploadContentType, text,
			map[string]any{"base64": base64.StdEncoding.EncodeToString(text)}, okMultipart(textPath, textUploadContentType, len(text))},
		{"multipart_empty", "an empty payload: still one multipart upload (ChunkSize 1)", "e/empty.bin", "application/octet-stream", []byte{},
			map[string]any{"base64": ""}, okMultipart("e/empty.bin", "application/octet-stream", 0)},
		{"multipart_largest", "16 MiB - 1 bytes: the largest multipart upload (ChunkSize 16 MiB)", binPath, "application/octet-stream", zzPattern(mib16 - 1),
			map[string]any{"pattern": "i % 251", "len": mib16 - 1}, okMultipart(binPath, "application/octet-stream", mib16-1)},
		{"resumable_exact_16mib", "exactly 16 MiB (googleapi.DefaultUploadChunkSize): a resumable upload in 16 MiB chunks", binPath, "application/octet-stream", zzPattern(mib16),
			map[string]any{"pattern": "i % 251", "len": mib16}, resumable(binPath, "application/octet-stream", nil)},
		{"resumable_over_16mib", "16 MiB + 1 bytes: a resumable upload, a full 16 MiB chunk then a 1-byte final chunk", binPath, "application/octet-stream", zzPattern(mib16 + 1),
			map[string]any{"pattern": "i % 251", "len": mib16 + 1}, resumable(binPath, "application/octet-stream", nil)},
		{"resumable_chunk_503_retried", "16 MiB + 1 bytes whose first chunk PUT is answered 503 once: the chunk is resent", binPath, "application/octet-stream", zzPattern(mib16 + 1),
			map[string]any{"pattern": "i % 251", "len": mib16 + 1}, resumable(binPath, "application/octet-stream", map[int]int{1: 503})},
		{"precondition_412", "the object exists: 412 conditionNotMet counts as success", pngPath, "image/png", small,
			map[string]any{"base64": base64.StdEncoding.EncodeToString(small)}, seq(zzStorageError(412, "conditionNotMet"))},
		{"retry_503", "a 503 then success: the idempotent create-only upload is retried with the same idempotency token", pngPath, "image/png", small,
			map[string]any{"base64": base64.StdEncoding.EncodeToString(small)}, seq(zzStorageError(503, "backendError"), zzObjectReply(pngPath, len(small), "image/png"))},
		{"retry_429_then_500", "429 then 500 then success: both retried", pngPath, "image/png", small,
			map[string]any{"base64": base64.StdEncoding.EncodeToString(small)}, seq(zzStorageError(429, "rateLimitExceeded"), zzStorageError(500, "backendError"), zzObjectReply(pngPath, len(small), "image/png"))},
		{"error_403", "403 forbidden: not retried, returned as the owned upload error", pngPath, "image/png", small,
			map[string]any{"base64": base64.StdEncoding.EncodeToString(small)}, seq(zzStorageError(403, "forbidden"))},
		{"error_400", "400 badRequest: not retried", "p.txt", "text/plain", []byte("PAYLOAD-MARKER"),
			map[string]any{"base64": base64.StdEncoding.EncodeToString([]byte("PAYLOAD-MARKER"))}, seq(zzStorageError(400, "badRequest"))},
	}
	for _, c := range cases {
		s := &zzGCSServer{t: p.t, tokens: map[string]string{}, invIDs: map[string]string{}}
		s.srv = httptest.NewServer(http.HandlerFunc(s.handle))
		s.reply = c.reply(s)
		ctx := context.Background()
		o, err := newGCSOffloader(ctx, "test-bucket", option.WithEndpoint(s.srv.URL+"/storage/v1/"), option.WithoutAuthentication())
		if err != nil {
			p.t.Fatalf("%s: newGCSOffloader: %v", c.name, err)
		}
		uri, upErr := o.UploadContent(ctx, GCSUploadRequest{Path: c.path, Data: c.data, ContentType: c.contentType})
		closeErr := o.Close()
		s.srv.Close()
		var apiCode any
		var apiErr *googleapi.Error
		if errors.As(upErr, &apiErr) {
			apiCode = apiErr.Code
		}
		var owned ownedUploadError
		errText := any(nil)
		if upErr != nil {
			errText = s.template(upErr.Error())
		}
		crc := crc32.Checksum(c.data, crc32.MakeTable(crc32.Castagnoli))
		var crcBE [4]byte
		binary.BigEndian.PutUint32(crcBE[:], crc)
		data := map[string]any{"len": len(c.data), "sha256": zzSHA256(c.data), "crc32c": crc, "crc32c_base64": base64.StdEncoding.EncodeToString(crcBE[:])}
		for k, v := range c.dataDesc {
			data[k] = v
		}
		p.writeJSON("gcs/"+c.name+".json", map[string]any{
			"name": c.name, "description": c.description,
			"client":    map[string]any{"bucket": "test-bucket", "endpoint": "{{SERVER}}/storage/v1/", "options": "option.WithEndpoint(server+\"/storage/v1/\"), option.WithoutAuthentication()"},
			"call":      map[string]any{"path": c.path, "content_type": c.contentType, "data": data},
			"exchanges": s.exchanges,
			"result": map[string]any{
				"uri": uri, "error": errText, "error_type": zzErrType(upErr), "is_owned_upload_error": errors.As(upErr, &owned),
				"googleapi_code": apiCode, "is_precondition_failed": isPreconditionFailed(upErr),
				"safe_upload_error_summary": zzSummaryOrNil(upErr), "close_error": zzErrText(closeErr),
			},
		})
	}
}

func zzSummaryOrNil(err error) any {
	if err == nil {
		return nil
	}
	return safeUploadErrorSummary(err)
}

func (p *zzProbe) genHelpers() {
	// CRC32C.
	castagnoli := crc32.MakeTable(crc32.Castagnoli)
	ascending := make([]byte, 32)
	for i := range ascending {
		ascending[i] = byte(i)
	}
	type crcCase struct {
		name string
		data []byte
	}
	crcCases := []crcCase{
		{"empty", nil},
		{"a", []byte("a")},
		{"abc", []byte("abc")},
		{"check_123456789", []byte("123456789")},
		{"zeros_32", make([]byte, 32)},
		{"ones_32", bytes.Repeat([]byte{0xff}, 32)},
		{"ascending_32", ascending},
		{"png_7", []byte{0x89, 'P', 'N', 'G', 0, 1, 2}},
		{"utf8_text", []byte("héllo, offloaded text ✓")},
		{"pattern_1mib", zzPattern(1 << 20)},
		{"pattern_16mib", zzPattern(16 << 20)},
	}
	var crcOut []any
	for _, c := range crcCases {
		v := crc32.Checksum(c.data, castagnoli)
		var be [4]byte
		binary.BigEndian.PutUint32(be[:], v)
		entry := map[string]any{"name": c.name, "len": len(c.data), "crc32c": v, "hex": fmt.Sprintf("%08x", v), "base64_be": base64.StdEncoding.EncodeToString(be[:])}
		if len(c.data) <= 64 {
			entry["data_base64"] = base64.StdEncoding.EncodeToString(c.data)
		} else {
			entry["data_pattern"] = "i % 251"
		}
		// Incremental update over two halves equals the one-shot value.
		h := crc32.Update(crc32.Update(0, castagnoli, c.data[:len(c.data)/2]), castagnoli, c.data[len(c.data)/2:])
		if h != v {
			p.t.Fatalf("crc32c %s: incremental %08x != %08x", c.name, h, v)
		}
		crcOut = append(crcOut, entry)
	}
	p.writeCases("helpers/crc32c.json",
		"crc32.Checksum(data, crc32.MakeTable(crc32.Castagnoli)) (crc32c as uint32 and hex) and base64_be = std base64 of its 4 big-endian bytes (the crc32c field of the Cloud Storage object metadata). data is data_base64, or data_pattern \"i % 251\" (byte i = i mod 251) of len bytes.",
		crcOut)

	p.writeCases("helpers/extension_for_mime.json",

	// safeUploadErrorSummary.
	apiErr := &googleapi.Error{Code: 403, Message: "forbidden https://storage.googleapis.com/b/o?token=abc"}
	type sumCase struct {
